include(MmltkComponent)
find_package(PkgConfig REQUIRED)
pkg_check_modules(MMLTK_FFMPEG REQUIRED IMPORTED_TARGET libavformat libavcodec libavutil libswscale)
add_library(mmltk_backend_media_video STATIC)
add_library(mmltk::backend::media::video ALIAS mmltk_backend_media_video)
mmltk_configure_component(mmltk_backend_media_video PUBLIC_DECLARATIONS)
mmltk_register_public_headers(mmltk_backend_media_video BASE_DIRS ${CMAKE_SOURCE_DIR} FILES video_file_source.h)
mmltk_register_private_headers(mmltk_backend_media_video video_frame_convert.h)
mmltk_register_retained_module_importers(mmltk_backend_media_video video_file_source.cpp)
mmltk_register_ordinary_sources(mmltk_backend_media_video video_frame_convert.cu)
target_link_libraries(mmltk_backend_media_video PRIVATE PkgConfig::MMLTK_FFMPEG CUDA::cudart CUDA::cuda_driver mmltk::frameworks::gpu mmltk::common::logging)
mmltk_target_cuda_architectures(mmltk_backend_media_video)
mmltk_register_header_isolation(mmltk_backend_media_video)
function(mmltk_add_backend_media_video_tests)
    add_executable(mmltk_backend_media_video_tests EXCLUDE_FROM_ALL)
    mmltk_configure_component(mmltk_backend_media_video_tests PRIVATE_DECLARATIONS)
    mmltk_register_ordinary_sources(mmltk_backend_media_video_tests ${CMAKE_CURRENT_FUNCTION_LIST_DIR}/tests/video_file_source.test.cpp)
    target_link_libraries(mmltk_backend_media_video_tests PRIVATE mmltk_acceptance_support PkgConfig::MMLTK_FFMPEG mmltk::backend::media::video CUDA::cudart mmltk::frameworks::gpu mmltk::common::system)
    mmltk_enable_native_runtime_rpath(mmltk_backend_media_video_tests)
    catch_discover_tests(mmltk_backend_media_video_tests TEST_PREFIX "backend::media::video::" DISCOVERY_MODE PRE_TEST ADD_TAGS_AS_LABELS)
endfunction()
